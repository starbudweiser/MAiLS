        public static let bgSurfaceElevated = adaptiveColor(light: 0xF0F2F5, dark: 0x262930)

        public static let bgSelected = adaptiveColor(light: 0xE5E9F0, dark: 0x2B303B)

        public static let borderStrong = adaptiveColor(light: 0xCBD2DF, dark: 0x3B404C)

        public static let textPrimary = adaptiveColor(light: 0x111827, dark: 0xF2F4F8)
        public static let textSecondary = adaptiveColor(light: 0x4B5563, dark: 0x9CA3AF)
        public static let textMuted = adaptiveColor(light: 0x9CA3AF, dark: 0x6B7280)

        public static let accentPrimary = adaptiveColor(light: 0xEA580C, dark: 0xFF6B2B)
        public static let accentPrimaryHover = adaptiveColor(light: 0xC2410C, dark: 0xE85A1C)

        public static let nsBgList = adaptiveNSColor(light: 0xFAFAFB, dark: 0x18191D)
        public static let nsBgHover = adaptiveNSColor(light: 0xEFEFEF, dark: 0x23262E)
        public static let nsBgSelected = adaptiveNSColor(light: 0xE5E9F0, dark: 0x2B303B)
        public static let nsBorderSubtle = adaptiveNSColor(light: 0xE2E5EC, dark: 0x2A2D35)
        public static let nsTextPrimary = adaptiveNSColor(light: 0x111827, dark: 0xF2F4F8)
        public static let nsTextSecondary = adaptiveNSColor(light: 0x4B5563, dark: 0x9CA3AF)
        public static let nsTextMuted = adaptiveNSColor(light: 0x9CA3AF, dark: 0x6B7280)
        public static let nsAccentPrimary = adaptiveNSColor(light: 0xEA580C, dark: 0xFF6B2B)

        private static func adaptiveColor(light: UInt32, dark: UInt32) -> Color {
            Color(nsColor: adaptiveNSColor(light: light, dark: dark))
        }

        private static func adaptiveNSColor(light: UInt32, dark: UInt32) -> NSColor {
            NSColor(name: nil) { appearance in
                let isDark = appearance.bestMatch(from: [.darkAqua, .aqua]) == .darkAqua
                return nsColor(from: isDark ? dark : light)
            }
        }

        private static func nsColor(from hex: UInt32) -> NSColor {
            let r = CGFloat((hex >> 16) & 0xFF) / 255.0
            let g = CGFloat((hex >> 8) & 0xFF) / 255.0
            let b = CGFloat(hex & 0xFF) / 255.0
            return NSColor(srgbRed: r, green: g, blue: b, alpha: 1.0)
        }
    }
